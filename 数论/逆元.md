```c++
// exgcd 求逆元

int inv_i(int a, int m) {
    int x, y;
    int d = exgcd(a, m, x, y);
    // d==1 时逆元存在，x即为逆元
    return (x % m + m) % m; // 调整为正数
}
```

```c++
// 预处理逆元加逆元的阶乘
int fac[N], inv[N];
void init(int p) {
    fac[0] = 1;
    for (int i = 1;i<p;i++) fac[i] = fac[i-1]*i%p;
    inv[p-1] = q_pow(fac[p-1],p-2,p);
    for (int i = 2;i<=n;i++) {
        inv[i] = (p-p/i)*inv[p%i]%p;// 线性递推求逆元
    }
    for (int i = 1;i<=n;i++) {
        inv[i] = inv[i-1]*inv[i]%p;
    }
}
```

```c++
int C(int n,int m,int p) {
```

```c++
// Lucas 定理主函数
int lucas(int n,int m,int p) {
    if (m == 0) return 1;
    return C(n % p, m % p, p) * lucas(n / p, m / p, p) % p;
}
```

```c++
// 快速幂
int q_pow(int a,int b) {
    int res = 1;
    while (b) {
        if (b&1) {
            res *= a;
            res %= p;
        }
        a*=a;
        a%=p;
        b>>=1;
    }
    return res;
}
```
